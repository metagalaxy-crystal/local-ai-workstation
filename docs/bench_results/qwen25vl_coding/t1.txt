def merge(intervals):
    if not intervals:
        return []
    
    intervals.sort(key=lambda x: x[0])
    merged = [intervals[0]]
    
    for interval in intervals[1:]:
        last_merged = merged[-1]
        if last_merged[1] >= interval[0]:
            merged[-1] = [last_merged[0], max(last_merged[1], interval[1])]
        else:
            merged.append(interval)
    
    return merged
