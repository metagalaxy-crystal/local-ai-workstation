class LRUCache:
    def __init__(self, capacity: int):
        self.cache = {}
        self.capacity = capacity
        self.link = {key: [self, self]}  # (next, prev) for doubly linked list

    def get(self, key: int) -> int:
        if key in self.cache:
            node = self.cache[key]
            self.link[node[0]].pop(0)  # Move to front
            return self.cache[key]
        return -1

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache[key].pop(0)  # Remove and re-add
        self.cache[key] = (key, value)
        if len(self.cache) > self.capacity:
            del self.cache[self.link[self.cache[-1]][0]]
        self.link[self.cache[-1]].append((key, value))
        self.capacity -= 1
